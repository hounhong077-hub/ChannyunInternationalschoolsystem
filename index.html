<script>

  // ---- Your deployed Apps Script /exec URL ----

  const APP_URL = "https://script.google.com/macros/s/AKfycbyGE1k0CWBfOxEUeI6_aXyzDxlcCauIFrbT3s6MfQtrxh8JBZsRCa-NJFMkhBfM0FtO/exec";


  const landing   = document.getElementById("landing");

  const dashboard = document.getElementById("dashboard");

  const frame     = document.getElementById("appFrame");

  const openBtn   = document.getElementById("openBtn");

  const backBtn   = document.getElementById("backBtn");


  openBtn.addEventListener("click", () => {

    // Load the Apps Script app only when the user chooses

    if (!frame.src) {

      frame.src = APP_URL;

    }

    landing.style.display = "none";

    dashboard.classList.add("active");

    dashboard.setAttribute("aria-hidden", "false");

    // Optional: update URL hash so refresh stays on dashboard

    history.pushState({ view: "dashboard" }, "", "#dashboard");

  });


  backBtn.addEventListener("click", () => {

    dashboard.classList.remove("active");

    dashboard.setAttribute("aria-hidden", "true");

    landing.style.display = "";

    history.pushState({ view: "landing" }, "", "#");

  });


  // Handle browser back/forward buttons

  window.addEventListener("popstate", (e) => {

    if (e.state && e.state.view === "dashboard") {

      openBtn.click();

    } else {

      dashboard.classList.remove("active");

      landing.style.display = "";

    }

  });


  // If the page was opened directly with #dashboard, show it

  if (location.hash === "#dashboard") {

    openBtn.click();

  }

</script>
